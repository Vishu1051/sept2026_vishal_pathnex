     ## ansible - deploy  app using git  





    ---
    - name: Deploy app from git  for Pathnex
      hosts: all 
      become: yes
      
      tasks:
        - name: Install git 
          yum:
            name: git 
            state: present
          



        - name: clone repository
          git:
             repo: "https://github.com/pathnex/app.git"
             dest: "/opt/pathnex-app"

   ## kubernetes - deployment rolling update 
     

   apiVersion: apps/v1 
   kind:  Deployment
   metadata: 
      name: pathnex-rollout 
    spec: 
       replicas: 3
       strategy:
           type: RollingUpdate
           rollingUpdate:
              maxSurge: 2
              maxUnavailable: 1
           

         selector:
            matchlevels:
               app: rollout
         template:
             metadata:
               labels:
                 app: rollout
           spec:
              containers:
               - name: app
                 image: nginx
