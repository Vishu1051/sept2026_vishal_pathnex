 ##  ansible - create  multiple users in loop

 
  ---
  - name: Create  multiple Pathnex users

    hosts: all
    become: yes
    
   
    vars:
      users:
        - dev
        - sam
        - doctor
   
    tasks: 
      - name: Create users
        users:
           name: "{{item}}"
           state: present
         loop: "{{users}}"
