             requiredDuringSchedulingIgnoredDuringExecution:

                 - matchExpressions:

  ##  jenkins pipeline - rollback on failure
        


    pipeline {
         any agent  
         enviroment {
             INSTITUTE_NAME = "Pathnex"
        }
         stages {
               stage('Deploy') {
                     steps {
                         script {
                            try {
                                sh 'echo "Deploying $INSTIUTE_NAME application..."'
                                sh 'exit 1' //  simulate failure 
              }       cath (Expection e) {
                        echo "Deployment failure. Rolling back $INSTITUTE_NAME application..."
                                      }
                                    }
                                  }
                          }
                   }
              }
         ##  gitlab ci  rollback  simulation 




            stages: 
              - deploy 
          

           variables:
             INSTITUTE_NAME: "Pathnex"
        
           deploy:
               stage: deploy
               image: alpine:latest
               script:
                 - echo "Deploying $INSTITUTE_NAME application..."
                 - exit 1 
               after_script:
                 - echo "Deployment failed. Rolling back $INSTITUTE_NAME application..."
